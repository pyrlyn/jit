//! x86-64 Assembler: direct emission into a CodeBuffer, no IR, no dispatch tables.
//! Encodings follow the Intel SDM; where AsmJit picks a shorter form (e.g. mov r64, u32
//! => mov r32, imm32) we pick the same one, so outputs are byte-identical.
const std = @import("std");
const core = @import("../core/code_buffer.zig");
const CodeBuffer = core.CodeBuffer;
pub const Label = core.Label;
pub const Error = core.Error;

pub const Gp = enum(u4) {
    rax, rcx, rdx, rbx, rsp, rbp, rsi, rdi, r8, r9, r10, r11, r12, r13, r14, r15,
    inline fn lo(r: Gp) u8 { return @intFromEnum(r) & 7; }
    inline fn hi(r: Gp) u8 { return @intFromEnum(r) >> 3; }
};

pub const Scale = enum(u2) { x1, x2, x4, x8 };

/// [base + index*scale + disp]. `base == null` => absolute/[index*scale+disp32].
pub const Mem = struct {
    base: ?Gp = null,
    index: ?Gp = null,
    scale: Scale = .x1,
    disp: i32 = 0,
};

pub inline fn ptr(base: Gp, disp: i32) Mem { return .{ .base = base, .disp = disp }; }
pub inline fn ptrIdx(base: Gp, index: Gp, scale: Scale, disp: i32) Mem {
    return .{ .base = base, .index = index, .scale = scale, .disp = disp };
}

pub const Cond = enum(u4) { o, no, b, ae, e, ne, be, a, s, ns, p, np, l, ge, le, g };

/// Classic two-operand ALU group: `op /digit` for imm forms, `op*8+1` (r/m,r) and `+3` (r,r/m).
pub const Alu = enum(u3) { add = 0, @"or" = 1, adc = 2, sbb = 3, @"and" = 4, sub = 5, xor = 6, cmp = 7 };

const REX_W: u8 = 0x48;

inline fn fitsI8(v: i64) bool { return v >= -128 and v <= 127; }
inline fn fitsI32(v: i64) bool { return v >= std.math.minInt(i32) and v <= std.math.maxInt(i32); }

pub const Assembler = struct {
    buf: *CodeBuffer,

    pub fn init(buf: *CodeBuffer) Assembler { return .{ .buf = buf }; }

    pub inline fn newLabel(a: Assembler) Error!Label { return a.buf.newLabel(); }
    pub inline fn bind(a: Assembler, l: Label) Error!void { return a.buf.bind(l); }

    // ---- encoding helpers (unchecked: caller reserved 16 bytes) -----------------

    inline fn rexRR(a: Assembler, w: bool, reg: u8, rm: Gp) void {
        const rex: u8 = (if (w) REX_W else 0x40) | (reg >> 3) << 2 | rm.hi();
        if (rex != 0x40) a.buf.put8(rex);
    }

    inline fn rexRM(a: Assembler, w: bool, reg: u8, m: Mem) void {
        const x: u8 = if (m.index) |i| i.hi() else 0;
        const b: u8 = if (m.base) |r| r.hi() else 0;
        const rex: u8 = (if (w) REX_W else 0x40) | (reg >> 3) << 2 | x << 1 | b;
        if (rex != 0x40) a.buf.put8(rex);
    }

    inline fn modrmRR(a: Assembler, reg: u8, rm: Gp) void {
        a.buf.put8(0xC0 | (reg & 7) << 3 | rm.lo());
    }

    /// ModRM [+SIB] [+disp] for a memory operand. `reg` is the /r or /digit field.
    fn modrmM(a: Assembler, reg: u8, m: Mem) void {
        const r: u8 = (reg & 7) << 3;
        const base = m.base orelse {
            // No base: SIB with base=101 and disp32 (index=100 means none).
            a.buf.put8(0x04 | r);
            const idx: u8 = if (m.index) |i| i.lo() else 4;
            a.buf.put8(@as(u8, @intFromEnum(m.scale)) << 6 | idx << 3 | 5);
            a.buf.put32(@bitCast(m.disp));
            return;
        };
        // rbp/r13 as base cannot use mod=00 (that encodes disp32/RIP).
        const mod: u8 = if (m.disp == 0 and base.lo() != 5) 0x00 else if (fitsI8(m.disp)) 0x40 else 0x80;
        if (m.index != null or base.lo() == 4) {
            const idx: u8 = if (m.index) |i| i.lo() else 4;
            a.buf.put8(mod | r | 4);
            a.buf.put8(@as(u8, @intFromEnum(m.scale)) << 6 | idx << 3 | base.lo());
        } else {
            a.buf.put8(mod | r | base.lo());
        }
        switch (mod) {
            0x40 => a.buf.put8(@bitCast(@as(i8, @intCast(m.disp)))),
            0x80 => a.buf.put32(@bitCast(m.disp)),
            else => {},
        }
    }

    inline fn opRR(a: Assembler, op: u8, reg: Gp, rm: Gp) Error!void {
        try a.buf.ensure(16);
        a.rexRR(true, @intFromEnum(reg), rm);
        a.buf.put8(op);
        a.modrmRR(@intFromEnum(reg), rm);
    }

    inline fn opRM(a: Assembler, op: u8, reg: u8, m: Mem) Error!void {
        try a.buf.ensure(16);
        a.rexRM(true, reg, m);
        a.buf.put8(op);
        a.modrmM(reg, m);
    }

    // ---- mov ------------------------------------------------------------------

    pub fn movRR(a: Assembler, dst: Gp, src: Gp) Error!void { return a.opRR(0x89, src, dst); }
    pub fn movRM(a: Assembler, dst: Gp, src: Mem) Error!void { return a.opRM(0x8B, @intFromEnum(dst), src); }
    pub fn movMR(a: Assembler, dst: Mem, src: Gp) Error!void { return a.opRM(0x89, @intFromEnum(src), dst); }

    pub fn movMI(a: Assembler, dst: Mem, imm: i32) Error!void {
        try a.opRM(0xC7, 0, dst);
        a.buf.put32(@bitCast(imm));
    }

    /// Picks the shortest encoding: u32 => `mov r32, imm32` (zero-extends),
    /// i32 => `mov r/m64, imm32` (sign-extends), otherwise `movabs r64, imm64`.
    pub fn movRI(a: Assembler, dst: Gp, imm: i64) Error!void {
        try a.buf.ensure(16);
        if (imm >= 0 and imm <= std.math.maxInt(u32)) {
            if (dst.hi() != 0) a.buf.put8(0x41);
            a.buf.put8(0xB8 + dst.lo());
            a.buf.put32(@intCast(imm));
        } else if (fitsI32(imm)) {
            a.rexRR(true, 0, dst);
            a.buf.put8(0xC7);
            a.modrmRR(0, dst);
            a.buf.put32(@bitCast(@as(i32, @intCast(imm))));
        } else {
            a.buf.put8(REX_W | dst.hi());
            a.buf.put8(0xB8 + dst.lo());
            a.buf.put64(@bitCast(imm));
        }
    }

    /// Always emits the 10-byte `movabs r64, imm64` (useful for patchable constants).
    pub fn movabs(a: Assembler, dst: Gp, imm: u64) Error!void {
        try a.buf.ensure(16);
        a.buf.put8(REX_W | dst.hi());
        a.buf.put8(0xB8 + dst.lo());
        a.buf.put64(imm);
    }

    pub fn lea(a: Assembler, dst: Gp, src: Mem) Error!void { return a.opRM(0x8D, @intFromEnum(dst), src); }

    // ---- ALU --------------------------------------------------------------------

    pub fn aluRR(a: Assembler, comptime op: Alu, dst: Gp, src: Gp) Error!void {
        return a.opRR(@as(u8, @intFromEnum(op)) * 8 + 1, src, dst);
    }
    pub fn aluRM(a: Assembler, comptime op: Alu, dst: Gp, src: Mem) Error!void {
        return a.opRM(@as(u8, @intFromEnum(op)) * 8 + 3, @intFromEnum(dst), src);
    }
    pub fn aluMR(a: Assembler, comptime op: Alu, dst: Mem, src: Gp) Error!void {
        return a.opRM(@as(u8, @intFromEnum(op)) * 8 + 1, @intFromEnum(src), dst);
    }
    pub fn aluRI(a: Assembler, comptime op: Alu, dst: Gp, imm: i32) Error!void {
        try a.buf.ensure(16);
        a.rexRR(true, 0, dst);
        if (fitsI8(imm)) {
            a.buf.put8(0x83);
            a.modrmRR(@intFromEnum(op), dst);
            a.buf.put8(@bitCast(@as(i8, @intCast(imm))));
        } else if (dst == .rax) {
            a.buf.put8(@as(u8, @intFromEnum(op)) * 8 + 5); // accumulator short form
            a.buf.put32(@bitCast(imm));
        } else {
            a.buf.put8(0x81);
            a.modrmRR(@intFromEnum(op), dst);
            a.buf.put32(@bitCast(imm));
        }
    }

    pub inline fn addRR(a: Assembler, d: Gp, s: Gp) Error!void { return a.aluRR(.add, d, s); }
    pub inline fn addRI(a: Assembler, d: Gp, i: i32) Error!void { return a.aluRI(.add, d, i); }
    pub inline fn subRR(a: Assembler, d: Gp, s: Gp) Error!void { return a.aluRR(.sub, d, s); }
    pub inline fn subRI(a: Assembler, d: Gp, i: i32) Error!void { return a.aluRI(.sub, d, i); }
    pub inline fn cmpRR(a: Assembler, d: Gp, s: Gp) Error!void { return a.aluRR(.cmp, d, s); }
    pub inline fn cmpRI(a: Assembler, d: Gp, i: i32) Error!void { return a.aluRI(.cmp, d, i); }
    pub inline fn xorRR(a: Assembler, d: Gp, s: Gp) Error!void { return a.aluRR(.xor, d, s); }

    pub fn testRR(a: Assembler, d: Gp, s: Gp) Error!void { return a.opRR(0x85, s, d); }

    pub fn imulRR(a: Assembler, dst: Gp, src: Gp) Error!void {
        try a.buf.ensure(16);
        a.rexRR(true, @intFromEnum(dst), src);
        a.buf.put16(0xAF0F);
        a.modrmRR(@intFromEnum(dst), src);
    }
    pub fn imulRM(a: Assembler, dst: Gp, src: Mem) Error!void {
        try a.buf.ensure(16);
        a.rexRM(true, @intFromEnum(dst), src);
        a.buf.put16(0xAF0F);
        a.modrmM(@intFromEnum(dst), src);
    }
    pub fn imulRRI(a: Assembler, dst: Gp, src: Gp, imm: i32) Error!void {
        try a.buf.ensure(16);
        a.rexRR(true, @intFromEnum(dst), src);
        if (fitsI8(imm)) {
            a.buf.put8(0x6B);
            a.modrmRR(@intFromEnum(dst), src);
            a.buf.put8(@bitCast(@as(i8, @intCast(imm))));
        } else {
            a.buf.put8(0x69);
            a.modrmRR(@intFromEnum(dst), src);
            a.buf.put32(@bitCast(imm));
        }
    }

    // ---- stack / control flow -----------------------------------------------------

    pub fn push(a: Assembler, r: Gp) Error!void {
        try a.buf.ensure(2);
        if (r.hi() != 0) a.buf.put8(0x41);
        a.buf.put8(0x50 + r.lo());
    }
    pub fn pop(a: Assembler, r: Gp) Error!void {
        try a.buf.ensure(2);
        if (r.hi() != 0) a.buf.put8(0x41);
        a.buf.put8(0x58 + r.lo());
    }
    pub fn ret(a: Assembler) Error!void {
        try a.buf.ensure(1);
        a.buf.put8(0xC3);
    }
    pub fn nop(a: Assembler) Error!void {
        try a.buf.ensure(1);
        a.buf.put8(0x90);
    }

    /// Branch to a label. Bound (backward) labels get rel8 if it fits; unbound
    /// (forward) labels always get rel32 and a fixup — same policy as AsmJit.
    inline fn branch(a: Assembler, comptime short_op: []const u8, comptime long_op: []const u8, l: Label) Error!void {
        try a.buf.ensure(8);
        const start: i64 = a.buf.offset();
        if (a.buf.labelOffset(l)) |target| {
            const d8 = @as(i64, target) - (start + short_op.len + 1);
            if (short_op.len != 0 and fitsI8(d8)) {
                inline for (short_op) |b| a.buf.put8(b);
                a.buf.put8(@bitCast(@as(i8, @intCast(d8))));
                return;
            }
            inline for (long_op) |b| a.buf.put8(b);
            a.buf.put32(@bitCast(@as(i32, @intCast(@as(i64, target) - (start + long_op.len + 4)))));
            return;
        }
        inline for (long_op) |b| a.buf.put8(b);
        const at = a.buf.offset();
        a.buf.put32(0);
        try a.buf.addFixup(.{ .at = at, .base = at + 4, .label = l.id, .kind = .rel32 });
    }

    pub fn jmp(a: Assembler, l: Label) Error!void { return a.branch(&.{0xEB}, &.{0xE9}, l); }
    pub fn call(a: Assembler, l: Label) Error!void { return a.branch(&.{}, &.{0xE8}, l); }
    pub fn jcc(a: Assembler, comptime c: Cond, l: Label) Error!void {
        const cc: u8 = @intFromEnum(c);
        return a.branch(&.{0x70 + cc}, &.{ 0x0F, 0x80 + cc }, l);
    }

    pub fn callR(a: Assembler, r: Gp) Error!void {
        try a.buf.ensure(3);
        if (r.hi() != 0) a.buf.put8(0x41);
        a.buf.put8(0xFF);
        a.modrmRR(2, r);
    }
    pub fn jmpR(a: Assembler, r: Gp) Error!void {
        try a.buf.ensure(3);
        if (r.hi() != 0) a.buf.put8(0x41);
        a.buf.put8(0xFF);
        a.modrmRR(4, r);
    }

    /// Calls an absolute address: `mov r11, imm64; call r11` (r11 is caller-saved scratch).
    pub fn callAbs(a: Assembler, addr: usize) Error!void {
        try a.movRI(.r11, @bitCast(@as(u64, addr)));
        try a.callR(.r11);
    }
};

// ---- tests: encodings verified against GNU as / AsmJit output ---------------------

fn expectEnc(expected: []const u8, comptime f: anytype, args: anytype) !void {
    var cb = CodeBuffer.init(std.testing.allocator);
    defer cb.deinit();
    const a = Assembler.init(&cb);
    try @call(.auto, f, .{a} ++ args);
    try std.testing.expectEqualSlices(u8, expected, cb.code());
}

test "mov" {
    try expectEnc(&.{ 0x48, 0x89, 0xC8 }, Assembler.movRR, .{ Gp.rax, Gp.rcx }); // mov rax, rcx
    try expectEnc(&.{ 0x4D, 0x89, 0xC8 }, Assembler.movRR, .{ Gp.r8, Gp.r9 }); // mov r8, r9
    try expectEnc(&.{ 0xB8, 42, 0, 0, 0 }, Assembler.movRI, .{ Gp.rax, @as(i64, 42) }); // mov eax, 42
    try expectEnc(&.{ 0x41, 0xBB, 1, 0, 0, 0 }, Assembler.movRI, .{ Gp.r11, @as(i64, 1) }); // mov r11d, 1
    try expectEnc(&.{ 0x48, 0xC7, 0xC1, 0xFF, 0xFF, 0xFF, 0xFF }, Assembler.movRI, .{ Gp.rcx, @as(i64, -1) }); // mov rcx, -1
    try expectEnc(&.{ 0x48, 0xB8, 0xEF, 0xCD, 0xAB, 0x89, 0x67, 0x45, 0x23, 0x01 }, Assembler.movRI, .{ Gp.rax, @as(i64, 0x0123456789ABCDEF) });
    try expectEnc(&.{ 0x49, 0xBF, 0, 0, 0, 0, 1, 0, 0, 0 }, Assembler.movRI, .{ Gp.r15, @as(i64, 1 << 32) });
}

test "mem operands" {
    try expectEnc(&.{ 0x48, 0x8B, 0x07 }, Assembler.movRM, .{ Gp.rax, ptr(.rdi, 0) }); // mov rax, [rdi]
    try expectEnc(&.{ 0x48, 0x8B, 0x04, 0x24 }, Assembler.movRM, .{ Gp.rax, ptr(.rsp, 0) }); // [rsp]
    try expectEnc(&.{ 0x48, 0x8B, 0x45, 0x00 }, Assembler.movRM, .{ Gp.rax, ptr(.rbp, 0) }); // [rbp]
    try expectEnc(&.{ 0x49, 0x8B, 0x45, 0x00 }, Assembler.movRM, .{ Gp.rax, ptr(.r13, 0) }); // [r13]
    try expectEnc(&.{ 0x49, 0x8B, 0x04, 0x24 }, Assembler.movRM, .{ Gp.rax, ptr(.r12, 0) }); // [r12]
    try expectEnc(&.{ 0x48, 0x8B, 0x47, 0x08 }, Assembler.movRM, .{ Gp.rax, ptr(.rdi, 8) }); // [rdi+8]
    try expectEnc(&.{ 0x48, 0x8B, 0x87, 0x00, 0x01, 0, 0 }, Assembler.movRM, .{ Gp.rax, ptr(.rdi, 256) });
    try expectEnc(&.{ 0x4A, 0x8B, 0x44, 0xC7, 0x10 }, Assembler.movRM, .{ Gp.rax, ptrIdx(.rdi, .r8, .x8, 16) }); // [rdi+r8*8+16]
    try expectEnc(&.{ 0x4C, 0x89, 0x4C, 0x24, 0xF8 }, Assembler.movMR, .{ ptr(.rsp, -8), Gp.r9 }); // mov [rsp-8], r9
    try expectEnc(&.{ 0x48, 0x8D, 0x04, 0x8E }, Assembler.lea, .{ Gp.rax, ptrIdx(.rsi, .rcx, .x4, 0) }); // lea rax,[rsi+rcx*4]
    try expectEnc(&.{ 0x48, 0x8B, 0x04, 0xCD, 0x10, 0, 0, 0 }, Assembler.movRM, .{ Gp.rax, Mem{ .index = .rcx, .scale = .x8, .disp = 16 } });
    try expectEnc(&.{ 0x48, 0xC7, 0x07, 5, 0, 0, 0 }, Assembler.movMI, .{ ptr(.rdi, 0), @as(i32, 5) }); // mov qword [rdi], 5
}

test "alu" {
    try expectEnc(&.{ 0x48, 0x01, 0xF7 }, Assembler.addRR, .{ Gp.rdi, Gp.rsi }); // add rdi, rsi
    try expectEnc(&.{ 0x48, 0x83, 0xC0, 0x01 }, Assembler.addRI, .{ Gp.rax, @as(i32, 1) }); // add rax, 1
    try expectEnc(&.{ 0x48, 0x05, 0x00, 0x01, 0, 0 }, Assembler.addRI, .{ Gp.rax, @as(i32, 256) }); // add rax, 256
    try expectEnc(&.{ 0x49, 0x81, 0xEA, 0x00, 0x01, 0, 0 }, Assembler.subRI, .{ Gp.r10, @as(i32, 256) }); // sub r10, 256
    try expectEnc(&.{ 0x4C, 0x29, 0xC1 }, Assembler.subRR, .{ Gp.rcx, Gp.r8 }); // sub rcx, r8
    try expectEnc(&.{ 0x48, 0x39, 0xF7 }, Assembler.cmpRR, .{ Gp.rdi, Gp.rsi }); // cmp rdi, rsi
    try expectEnc(&.{ 0x48, 0x0F, 0xAF, 0xC1 }, Assembler.imulRR, .{ Gp.rax, Gp.rcx }); // imul rax, rcx
    try expectEnc(&.{ 0x4D, 0x6B, 0xC1, 0x0A }, Assembler.imulRRI, .{ Gp.r8, Gp.r9, @as(i32, 10) }); // imul r8, r9, 10
    try expectEnc(&.{ 0x48, 0x03, 0x47, 0x08 }, Assembler.aluRM, .{ Alu.add, Gp.rax, ptr(.rdi, 8) }); // add rax,[rdi+8]
}

test "stack and calls" {
    try expectEnc(&.{0x55}, Assembler.push, .{Gp.rbp});
    try expectEnc(&.{ 0x41, 0x57 }, Assembler.push, .{Gp.r15});
    try expectEnc(&.{ 0x41, 0x5C }, Assembler.pop, .{Gp.r12});
    try expectEnc(&.{0xC3}, Assembler.ret, .{});
    try expectEnc(&.{ 0x41, 0xFF, 0xD3 }, Assembler.callR, .{Gp.r11}); // call r11
    try expectEnc(&.{ 0xFF, 0xD0 }, Assembler.callR, .{Gp.rax});
}

test "labels: forward rel32, backward rel8/rel32" {
    var cb = CodeBuffer.init(std.testing.allocator);
    defer cb.deinit();
    const a = Assembler.init(&cb);
    const top = try a.newLabel();
    const out = try a.newLabel();
    try a.bind(top);
    try a.jcc(.e, out); // 0F 84 rel32 (forward)
    try a.nop();
    try a.jmp(top); // EB rel8 backward: -(6+1+2) = -9
    try a.bind(out);
    try a.ret();
    try cb.finalize();
    try std.testing.expectEqualSlices(u8, &.{ 0x0F, 0x84, 0x03, 0, 0, 0, 0x90, 0xEB, 0xF7, 0xC3 }, cb.code());
}
